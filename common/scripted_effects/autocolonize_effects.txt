autocolonize_run = {
	# 年度执行总入口（autocolonize.5 的 body）：按当前 mode 分派

		limit = { has_country_flag = autocolonize_mode_continent }
		autocolonize_colonize_superregion
	}
	if = {
		limit = { has_country_flag = autocolonize_mode_trade_node }
		autocolonize_colonize_trade_node
	}
}

		}

country_event = {
	id = autocolonize.1
	title = autocolonize.1.title
	desc = autocolonize.1.desc
	is_triggered_only = yes
	trigger = {
		ai = no
		has_country_flag = autocolonize_active
	}
	option = {
		name = autocolonize.1.a
		custom_tooltip = autocolonize_tooltip_active
		set_country_flag = autocolonize_mode_continent
		clr_country_flag = autocolonize_mode_trade_node
		clr_country_flag = autocolonize_sort_distance
		clr_country_flag = autocolonize_sort_development
		country_event = { id = autocolonize.2 }
	}

	option = {
		name = autocolonize.1.b
		custom_tooltip = autocolonize_tooltip_active
		set_country_flag = autocolonize_mode_trade_node
		clr_country_flag = autocolonize_mode_continent
		clr_country_flag = autocolonize_sort_distance
		clr_country_flag = autocolonize_sort_development
		country_event = { id = autocolonize.2 }
	}

	option = {
		name = autocolonize.1.c
		autocolonize_reset_all
	}

}
